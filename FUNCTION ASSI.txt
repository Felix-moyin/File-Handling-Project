# ===============
# NUMBER GUESSING GAME
# ===============

import random

user_range1 = int(input("Enter the lower limit: "))
user_range2 = int(input("Enter the upper limit: "))

target = random.randint(user_range1, user_range2)
total_guess = 7
guess_count = 0

print(f"I have selected a number between {user_range1} and {user_range2}")

print()
print("Try to guess it! ")
print(f"You have just {total_guess} chances")

while True:
    print()
    guess = input(f"Guess the number: ")
    
    if not guess.isdigit():
        print("Invalid Input. ")
        print(f"You have {total_guess - guess_count} guesses left. ")
        
        continue

        
    guess = int(guess)
    guess_count += 1
    
    guesses_left = total_guess - guess_count 
    
    if guess > target: 
        print("The number too high! ")
    elif guess < target:
        print("The number is too low! ")
    elif guess == target:
        print("Get in jhoor! You guessed right. ")
        print(f"You got it in {guess_count} guesses. ")
        
        break
    print(f"You have {guesses_left} guesses left.  ")
    
    if guess_count  == total_guess:
        print()
        print(f"Game over! The number was {target}.")
        break
